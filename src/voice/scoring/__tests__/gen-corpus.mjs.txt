async function get(u: string, file: string): Promise<Buffer | null> {
  if (fs.existsSync(file)) return fs.statSync(file).size ? fs.readFileSync(file) : null
  for (let i = 0; i < 3; i++) {
    try { const r = await fetch(u); if (r.status === 404) break; if (r.ok) { const b = Buffer.from(await r.arrayBuffer()); fs.writeFileSync(file, b); return b } } catch {}
  }
  fs.writeFileSync(file, ''); return null
}
